
g++ -std=c++23 -Wall -Wextra -Wconversion -Wsign-conversion -pedantic-errors main.cpp -o main

g++                 compiler
-std=c++23          language standard
-Wall               warnings
-Wextra             more warnings
-Wconversion        conversion warnings
-Wsign-conversion   signed/unsigned conversion warnings
-pedantic-errors    reject non-standard extensions
main.cpp             source
-o main              output executable
